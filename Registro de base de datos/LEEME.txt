EVENTO ACTUALIZACIÓN
CREATE EVENT actualizar_prestamos_vencidos
   ON SCHEDULE EVERY 1 HOUR
   STARTS CURRENT_TIMESTAMP
   DO
   UPDATE prestamos
   SET estado = 'vencido'
   WHERE NOW() > fyh_devolucion
     AND (estado IS NULL OR estado != 'vencido');
